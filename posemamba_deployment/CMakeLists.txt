cmake_minimum_required(VERSION 3.22)
project(posemamba_scan_candidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
option(SCAN_BUILD_ORT "Build the ORT adapter (requires matching external headers)" OFF)
add_library(posemamba_scan SHARED src/scan.cpp)
target_include_directories(posemamba_scan PUBLIC include)
target_link_libraries(posemamba_scan PRIVATE Threads::Threads)
target_compile_options(posemamba_scan PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra)
if(SCAN_BUILD_ORT)
  if(NOT EXISTS "${ORT_INCLUDE_DIR}/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "Provide ORT_INCLUDE_DIR for the exact Android/host ORT version")
  endif()
  add_library(posemamba_scan_ort SHARED src/ort_adapter.cpp src/scan.cpp)
  target_include_directories(posemamba_scan_ort PRIVATE include "${ORT_INCLUDE_DIR}")
  target_link_libraries(posemamba_scan_ort PRIVATE Threads::Threads)
  target_compile_options(posemamba_scan_ort PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra)
endif()
add_executable(scan_smoke tests/smoke.cpp)
target_link_libraries(scan_smoke PRIVATE posemamba_scan)
add_executable(scan_benchmark src/benchmark.cpp)
target_link_libraries(scan_benchmark PRIVATE posemamba_scan)
enable_testing()
add_test(NAME scan_smoke COMMAND scan_smoke)
